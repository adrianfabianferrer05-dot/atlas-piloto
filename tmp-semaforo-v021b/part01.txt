//----------------------- utilidades de tiempo -----------------------
datetime MakeDT(int y,int m,int d,int hh=0,int mm=0,int ss=0)
  {
   MqlDateTime x={};
   x.year=y; x.mon=m; x.day=d; x.hour=hh; x.min=mm; x.sec=ss;
   return StructToTime(x);
  }

datetime DateOnly(datetime t)
  {
   MqlDateTime x={}; TimeToStruct(t,x);
   return MakeDT(x.year,x.mon,x.day,0,0,0);
  }

int NthSunday(int year,int mon,int nth)
  {
   datetime first=MakeDT(year,mon,1,0,0,0);
   MqlDateTime f={}; TimeToStruct(first,f);
   int add=(7-f.day_of_week)%7;
   return 1+add+7*(nth-1);
  }

bool IsNYDSTLocal(datetime ny)
  {
   MqlDateTime x={}; TimeToStruct(ny,x);
   int march_sun=NthSunday(x.year,3,2);
   int nov_sun=NthSunday(x.year,11,1);
   datetime start=MakeDT(x.year,3,march_sun,2,0,0);
   datetime end=MakeDT(x.year,11,nov_sun,2,0,0);
   return (ny>=start && ny<end);
  }

bool IsNYDSTUtc(datetime utc)
  {
   MqlDateTime x={}; TimeToStruct(utc,x);
   int march_sun=NthSunday(x.year,3,2);
   int nov_sun=NthSunday(x.year,11,1);
   datetime start_utc=MakeDT(x.year,3,march_sun,7,0,0); // 02:00 EST
   datetime end_utc=MakeDT(x.year,11,nov_sun,6,0,0);   // 02:00 EDT
   return (utc>=start_utc && utc<end_utc);
  }

datetime NYToUTC(datetime ny)
  {
   return ny + (IsNYDSTLocal(ny) ? 4*3600 : 5*3600);
  }

datetime UTCToNY(datetime utc)
  {
   return utc - (IsNYDSTUtc(utc) ? 4*3600 : 5*3600);
  }

datetime CurrentUTC()
  {
   MqlTick t={};
   if(SymbolInfoTick(g_gold,t) && t.time>0) return t.time;
   return TimeGMT();
  }

datetime TradingDateFromNY(datetime ny)
  {
   MqlDateTime x={}; TimeToStruct(ny,x);
   datetime d=MakeDT(x.year,x.mon,x.day,0,0,0);
   if(x.hour>=18) d+=86400;
   return d;
  }

void TradingBoundsUTC(datetime trading_date,datetime &start_utc,datetime &end_utc)
  {
   MqlDateTime d={}; TimeToStruct(trading_date,d);
   datetime start_ny=MakeDT(d.year,d.mon,d.day,18,0,0)-86400;
   datetime end_ny=MakeDT(d.year,d.mon,d.day,17,0,0);
   start_utc=NYToUTC(start_ny);
   end_utc=NYToUTC(end_ny);
  }

datetime NYAt(datetime trading_date,int hh,int mm=0)
  {
   MqlDateTime d={}; TimeToStruct(trading_date,d);
   return MakeDT(d.year,d.mon,d.day,hh,mm,0);
  }
